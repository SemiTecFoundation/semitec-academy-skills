\chapter{Աշխատանքային պլան և հանգրվաններ}
\label{ch:plan}

\begin{workplan}[Աշխատանքային պլան (մոտ 30 շաբաթ).]
  \milestone{1--3}{Մեկնարկ. միջավայր, ընթերցանություն, ներածական վարժություններ}{Վարժությունները ընդունված են}
  \milestone{4--5}{Միջերեսների համաձայնեցում, ճարտարապետության մանրամասնում}{Միջերեսի փաստաթուղթ, ռեգիստրների քարտեզ}
  \milestone{6--12}{Ու1. գեներատորներ, էմուլյատոր, FIFO + testbench-եր. Ու2. բազա, API, էմուլյատոր-սկրիպտ, dashboard}{Սիմուլյացիայի արդյունքներ; dashboard-ը աշխատում է էմուլյատորով}
  \milestone{13--18}{Ու1. processing IP, AXI ինտեգրում, PS ծրագիր. Ու2. վիզուալիզացիա, վիճակագրություն}{Տվյալները հարթակից հասնում են PS; փորձերի համեմատում}
  \milestone{19--22}{Ինտեգրում. FPGA $\leftrightarrow$ սերվեր, runtime կոնֆիգուրացիա}{Ամբողջական ցիկլը աշխատում է (բաժին~\ref{ch:demo}, քայլեր 1--6)}
  \milestone{23--25}{Անոմալիաների հայտնաբերում, չափումներ (timing, ռեսուրսներ, latency)}{Թեստերի և չափումների հաշվետվություն}
  \milestone{26--28}{Դիպլոմային աշխատանքների գրում}{Ամբողջական սևագրեր ղեկավարներին}
  \milestone{29--30}{Պաշտպանության նախապատրաստում}{Սլայդներ, ցուցադրման փորձ}
\end{workplan}

\notebox{Հանդիպումներ՝ շաբաթը մեկ անգամ (ընդհանուր)։ Յուրաքանչյուր հանդիպումից առաջ
ուսանողները ուղարկում են կարճ գրառում. ինչ է արված, ինչն է խոչընդոտում, հաջորդ քայլերը։}
